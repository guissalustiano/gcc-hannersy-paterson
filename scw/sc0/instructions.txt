# sc0: rv32i subset — lw sw beq add addi sub and or
# No jalr, lui, auipc, shifts, or logical immediates.
# Function calls are not supported (no PC-relative addressing).

# arithmetic
add
addi
sub
neg     # sub rd, x0, rs

# logic
and
or

# load / store
lw
sw

# branch
beq
beqz    # beq rs, x0, offset

# pseudo load / move
li      # addi rd, x0, imm  (small immediates only)
mv      # addi rd, rs, 0

# NOP
nop     # addi x0, x0, 0
